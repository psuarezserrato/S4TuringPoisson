\chapter{Status and conventions}
\label{chap_status}

This blueprint describes the Lean~4 formalization of P.~Su\'arez-Serrato, \emph{Hamiltonian and
Turing complete flows on smooth $4$-manifolds} (\texttt{S4TCdiffeo-Aug-2026.tex}), together with
the smoothing theorem of Cardona--Miranda--Peralta-Salas--Presas (CMPP21, Thm.~5.2) and the
Auroux--Donaldson--Katzarkov (ADK05) and Lekili (Lek09) wrinkled/broken Lefschetz fibration
machinery on which it depends. All nodes carrying \texttt{\textbackslash leanok} are proved in
Lean (toolchain \texttt{v4.36.0}, Mathlib and TauCeti \texttt{v4.36.0}) and depend only on
\texttt{propext}, \texttt{Classical.choice} and \texttt{Quot.sound}.

The main results on $S^4$ (\cref{thm_main_s4,thm_s4_short,thm_undecidable_s4,thm_r5,thm_main_s4_adk})
carry \emph{no} hypotheses and cover both the explicit definite-fold projection model and the
global ADK broken Lefschetz fibration on $S^4$ (\cref{thm_adk_s4}) together with Lekili's local
$\R^4\to\R^2$ normal-form moves and compactly supported broken-to-purely-wrinkled homotopy theorem
(\cref{thm_lekili_moves}), preserving both the aligned volume form and (via Moser conjugation,
\cref{thm_moser_volume}) the exact round volume form $\Omega_{S^4}$. The class
\texttt{HasCMPPCantorCirculationHamiltonians} is globally inhabited (\cref{thm_cmpp_instance}), and
every target open set in \texttt{TuringCompleteWitness} is certified by a computable enumerator of
rational open rectangles (\cref{def_turing_complete}). For general compact Hausdorff smooth
$4$-manifolds $(M,V_M)$, the literal statement of Corollary~2 is proved unconditionally with exact
preservation of $V_M$ itself (\cref{cor_paper_all}, via \cref{ext_fibration_existence,thm_moser_volume}),
while the full generically rank-$2$ Poisson Corollary~2 is proved over any positively co-oriented
\texttt{PurelyWrinkledFibration} or \texttt{BrokenLefschetzFibration}
(\cref{thm_cor2,thm_cor2_purely_wrinkled}). All blueprint nodes are formalized and proved (\texttt{100\%} green).

\chapter{Interfaces: Turing completeness and volume forms}
\label{chap_interfaces}

\begin{definition}[Finite Turing machines and halting with an output window]
\label{def_fintm0}
\lean{S4TuringPoisson.FinTM0, S4TuringPoisson.HaltsWithWindow}
\leanok
A \texttt{FinTM0} is a Mathlib \texttt{Turing.TM0} machine with finite alphabet $\Gamma$ and finite
state set $\Lambda$. $\mathrm{HaltsWithWindow}(M,t,k,w)$ says that $M$ started on tape $t$ reaches
a terminal configuration whose tape agrees with $w$ on the cells $[-k,k]$.
\end{definition}

\begin{definition}[Turing complete self-map]
\label{def_turing_complete}
\lean{S4TuringPoisson.RatBox2, S4TuringPoisson.evalRatBox2,
  S4TuringPoisson.TuringCompleteWitness, S4TuringPoisson.IsTuringComplete}
\leanok
\uses{def_fintm0}
$\varphi\colon X\to X$ is Turing complete if there is an open product embedding
$\mathrm{openChart}\colon P\times D\hookrightarrow X$ with disk slice $\mathrm{chart}(y)=\mathrm{openChart}(p_0,y)$,
points $p_{M,t}=\mathrm{chart}(c_{M,t})$ (depending only on $(M,t)$, with computable $2^{-m}$
rational approximations and injective forward orbits $n\mapsto\varphi^n(p_{M,t})$), and nonempty
open target tubes $U_{M,k,w}=\mathrm{openChart}(P\times\bigcup_j \mathrm{evalRatBox2}(\mathrm{enumBoxes}(k,w,j)))$
generated by a computable rational-box enumerator (depending only on $(M,k,w)$ and avoiding
non-terminal initial configurations) such that
$\exists n,\ \varphi^n(p_{M,t})\in U_{M,k,w} \iff \mathrm{HaltsWithWindow}(M,t,k,w)$.
\end{definition}

\begin{lemma}[Turing complete maps are non-periodic]
\label{lem_tc_ne_id}
\lean{S4TuringPoisson.IsTuringComplete.ne_id, S4TuringPoisson.IsTuringComplete.iterate_ne_id}
\leanok
\uses{def_turing_complete}
If $\varphi$ is Turing complete then $\varphi^p\neq\mathrm{id}$ for every $p>0$ (in particular
$\varphi\neq\mathrm{id}$).
\end{lemma}
\begin{proof}\leanok
Every encoded initial configuration has an injective infinite forward orbit.
\end{proof}

\begin{theorem}[Undecidability of orbit reachability]
\label{thm_undecidable_general}
\lean{S4TuringPoisson.IsTuringComplete.undecidable_orbit_reachability}
\leanok
\uses{def_turing_complete}
For a Turing complete $\varphi$, reachability of a fixed nonempty open set from computably
approximable points is recursively enumerable but not decidable.
\end{theorem}
\begin{proof}\leanok
Reduce the halting problem of Mathlib's partial recursive codes to orbit reachability, via a
universal halting machine.
\end{proof}

\begin{lemma}[Transfer along open embeddings]
\label{lem_transfer}
\lean{S4TuringPoisson.IsTuringComplete.transfer}
\leanok
\uses{def_turing_complete}
Turing completeness of a disk map transfers to any map that agrees with it on an open product
$B_0\times D$ embedded in a larger space.
\end{lemma}
\begin{proof}\leanok
Push the encoded points and open sets forward along the embedding.
\end{proof}

\begin{definition}[Positive volume form]
\label{def_volume}
\lean{S4TuringPoisson.PositiveVolumeForm}
\leanok
A nowhere-vanishing smooth top form together with its finite Borel measure, positive on nonempty
open sets, such that pullback invariance implies measure preservation.
\end{definition}

\begin{definition}[Area-preserving disk isotopies]
\label{def_disk_isotopy}
\lean{S4TuringPoisson.AreaPreservingDiskIsotopy,
  S4TuringPoisson.HasTuringCompleteAreaPreservingDiskIsotopy}
\leanok
\uses{def_turing_complete}
Smooth isotopies of $\R^2$ with unit Jacobian, supported in a ball of radius $r_0<1$; the class
packages one whose time-one map on the unit disk is Turing complete.
\end{definition}

\chapter{Poisson geometry}
\label{chap_poisson}

\begin{definition}[Bivectors, Poisson structures, Hamiltonian fields]
\label{def_poisson}
\lean{S4TuringPoisson.SmoothBivector, S4TuringPoisson.PoissonStructure,
  S4TuringPoisson.SmoothBivector.hamiltonianVectorField, S4TuringPoisson.SmoothBivector.IsPoissonMap,
  S4TuringPoisson.SmoothBivector.rankAt}
\leanok
A smooth skew anchor $\#_\pi\colon T^*M\to TM$; a Poisson structure is one whose bracket satisfies
the Jacobi identity. $X_h=-\#_\pi(dh)$, Poisson maps preserve brackets, and $\mathrm{rank}_x\pi$
is the dimension of the image of $\#_{\pi,x}$.
\end{definition}

\begin{definition}[Unimodularity]
\label{def_unimodular}
\lean{S4TuringPoisson.PoissonStructure.IsUnimodularWithVolume,
  S4TuringPoisson.PoissonStructure.IsUnimodularPoisson}
\leanok
\uses{def_poisson}
$\pi$ is unimodular with respect to $\Omega$ if $\mathcal L_{X_h}\Omega=0$ for every $h$.
\end{definition}

\begin{definition}[Jacobian Poisson structure and generic rank $2$]
\label{def_jacobian}
\lean{S4TuringPoisson.jacobianSmoothBivector, S4TuringPoisson.jacobianPoisson,
  S4TuringPoisson.jacobianPoissonOfPositiveVolume,
  S4TuringPoisson.PoissonStructure.IsGenericallyRankTwo}
\leanok
\uses{def_poisson, def_volume}
For $f\colon M^4\to N^2$, a volume form $\Omega$ and an area form $\mu_N$, the bracket defined by
$\{g,h\}\,\Omega = dg\wedge dh\wedge f^*\mu_N$; $\pi$ is generically rank $2$ when
$\{x\in M\mid\mathrm{rank}_x\pi=2\}$ is open and dense in $M$.
\end{definition}

\begin{lemma}[Interior product of Hamiltonian fields]
\label{lem_jacobian_interior}
\lean{S4TuringPoisson.interior_hamiltonianVectorField_jacobianSmoothBivector,
  S4TuringPoisson.funSMulₗ_bracket_jacobianSmoothBivector}
\leanok
\uses{def_jacobian}
$\iota_{X_h}\Omega = dh\wedge f^*\mu_N$, and the defining top-form identity of the bracket holds.
\end{lemma}
\begin{proof}\leanok
Pointwise linear algebra on the anchor built from $\Omega$.
\end{proof}

\chapter{Fibrations}
\label{chap_fibrations}

\begin{definition}[Local singular models and chirality]
\label{def_local_models}
\lean{S4TuringPoisson.IsIndefiniteFoldAt, S4TuringPoisson.IsRealLefschetzAt,
  S4TuringPoisson.IsChiralRealLefschetzAt, S4TuringPoisson.IsWrinkledCuspAt}
\leanok
Indefinite quadratic fold, achiral/general real Lefschetz, orientation-preserving chiral real
Lefschetz, and wrinkled cusp singularities, defined through smooth local charts.
\end{definition}

\begin{definition}[Regular-disk, broken Lefschetz, and purely wrinkled fibrations]
\label{def_wrinkled}
\lean{S4TuringPoisson.RegularDiskFibration, S4TuringPoisson.BrokenLefschetzFibration,
  S4TuringPoisson.PurelyWrinkledFibration}
\leanok
\uses{def_local_models}
A \texttt{RegularDiskFibration} is a smooth map $f\colon M\to N$ with a disk-shaped open region of
regular values over which $f$ is smoothly trivial with compact surface fiber (imposing no
critical-locus normal form). \texttt{BrokenLefschetzFibration} extends it by requiring every
critical point on $M$ to be a real Lefschetz point or an indefinite fold;
\texttt{PurelyWrinkledFibration} requires every critical point on $M$ to be an indefinite fold or a
wrinkled cusp.
\end{definition}

\begin{definition}[Explicit singular fibration of $S^4$]
\label{def_s4_fibration}
\lean{S4TuringPoisson.s4Fibration, S4TuringPoisson.s4RegularDiskFibration}
\leanok
\uses{def_wrinkled}
$f(x_0,\dots,x_4)=\sigma^{-1}(x_0,x_1)$, with $\sigma^{-1}$ the inverse stereographic projection
from the south pole, packaged as a \texttt{RegularDiskFibration} with fiber $S^2$.
\end{definition}

\begin{lemma}[Critical locus of $f$]
\label{lem_s4_crit}
\lean{S4TuringPoisson.contMDiff_s4Fibration, S4TuringPoisson.criticalPoints_s4Fibration,
  S4TuringPoisson.criticalPoints_s4Fibration_nonempty}
\leanok
\uses{def_s4_fibration}
$f$ is smooth and $\Crit(f)=\{x_2=x_3=x_4=0\}$, a nonempty circle with dense complement on which
the fibers collapse to points (a definite fold, proved unable to lift to
\texttt{BrokenLefschetzFibration} or \texttt{PurelyWrinkledFibration}).
\end{lemma}
\begin{proof}\leanok
Surjectivity of $df_x$ reduces to that of the planar projection restricted to $T_xS^4$.
\end{proof}

\begin{definition}[Regular-disk vs.\ wrinkled/broken Lefschetz fibration classes]
\label{def_admits}
\lean{S4TuringPoisson.AdmitsRegularDiskFibration,
  S4TuringPoisson.AdmitsWrinkledOrBrokenLefschetzFibration,
  S4TuringPoisson.AdmitsWrinkledOrBrokenLefschetzFibration.ofBrokenLefschetz,
  S4TuringPoisson.AdmitsWrinkledOrBrokenLefschetzFibration.ofPurelyWrinkled}
\leanok
\uses{def_wrinkled, def_volume}
\texttt{AdmitsRegularDiskFibration} packages an ambient volume form, a
\texttt{RegularDiskFibration} $M\to S^2$ with compact fiber $F$, an area form on $F$, and positive
co-orientation of the trivialization; \texttt{AdmitsWrinkledOrBrokenLefschetzFibration} extends it
by enforcing that every critical point on $M$ is an indefinite fold, wrinkled cusp, or real
Lefschetz point.
\end{definition}

\begin{lemma}[$S^4$ carries the regular-disk fibration structure]
\label{lem_s4_instance}
\lean{S4TuringPoisson.instAdmitsRegularDiskFibrationSphere4}
\leanok
\uses{def_admits, lem_s4_crit}
$S^4$ is an instance of \texttt{AdmitsRegularDiskFibration}, with fiber $S^2$ and the round volume forms.
\end{lemma}
\begin{proof}\leanok
Explicit trivialization over a stereographic disk and a sign computation for co-orientation.
\end{proof}

\begin{lemma}[Darboux disk charts on surfaces]
\label{lem_darboux}
\lean{S4TuringPoisson.SymplecticDiskChart, S4TuringPoisson.exists_symplecticDiskChart,
  S4TuringPoisson.SymplecticDiskChart.mpullback_embed_eq}
\leanok
Every nonempty symplectic surface has a disk chart pulling the area form back to a positive
multiple of $dy_0\wedge dy_1$.
\end{lemma}
\begin{proof}\leanok

\end{proof}

\begin{theorem}[Lekili local moves and compactly supported broken-to-wrinkled homotopy (Track~A)]
\label{thm_lekili_moves}
\lean{S4TuringPoisson.lekiliMove1, S4TuringPoisson.singularities_lekiliMove1,
  S4TuringPoisson.singularities_lekiliMove4_pos, S4TuringPoisson.singularities_lekiliMove4_neg,
  S4TuringPoisson.gate52_lekiliMove1_three_cusps,
  S4TuringPoisson.BrokenLefschetzFibration.exists_purelyWrinkledFibration}
\leanok
\uses{def_wrinkled}
Explicit smooth $1$-parameter families $\R^4\to\R^2$ model Lekili's Moves~1--4 with closed-form
indefinite-fold and wrinkled-cusp charts (including the swallowtail flip Move~4). Shrinking the
regular base disk and gluing compactly supported Move~1 bumps deforms every
\texttt{BrokenLefschetzFibration} $B$ on a compact Hausdorff $4$-manifold via a smooth global
homotopy $H\colon\R\times M\to\mathrm{Base}$ supported in a compact neighborhood $K$ of the real
Lefschetz points (disjoint from all regular and fold points) to a \texttt{PurelyWrinkledFibration} $P$.
\end{theorem}
\begin{proof}\leanok
Classify cusps by Weierstrass preparation ($d=3$) and indefinite folds by implicit chart
inversion ($d=2$), prove finite isolation of real Lefschetz critical points by compactness, shrink
\texttt{baseRegion} to half radius, and eliminate Lefschetz points via pairwise-disjoint compactly
supported homotopies $t\mapsto\mathrm{compactLefschetzWrinkle}(t s_0)$.
\end{proof}

\begin{theorem}[Global Auroux--Donaldson--Katzarkov fibration on $S^4$ (Track~A)]
\label{thm_adk_s4}
\lean{S4TuringPoisson.s4ADKFibration, S4TuringPoisson.s4BrokenLefschetzFibration,
  S4TuringPoisson.s4ADKTorusBrokenLefschetzFibration, S4TuringPoisson.s4PurelyWrinkledFibration,
  S4TuringPoisson.s4ADKNorthFiberDiffeomorph, S4TuringPoisson.s4ADKTorusFiberDiffeomorph,
  S4TuringPoisson.s4ADKAdmits}
\leanok
\uses{def_admits, def_s4_fibration}
There is an explicit global \texttt{BrokenLefschetzFibration} and
\texttt{PurelyWrinkledFibration} $f_{\mathrm{ADK}}\colon S^4\to S^2$ realizing the round-handle
decomposition of Auroux--Donaldson--Katzarkov, with critical set equal to the equatorial circle
$\{x_2=x_3=x_4=0\}$ of indefinite quadratic folds, embedded $C^\infty$ regular fibers
diffeomorphic to $S^2$ (north) and $T^2=S^1\times S^1$ (south), and explicit smooth bundle
trivializations over both hemispheres.
\end{theorem}
\begin{proof}\leanok
Normalize the $\mathrm{SO}(3)$-equivariant map to $S^2$, certify critical points and indefinite-fold
normal forms, equip regular level fibers with TauCeti's submanifold structure, and trivialize over
regular disks around both poles.
\end{proof}

\begin{theorem}[Local regular-disk fibration and Track~B separation on closed $4$-manifolds]
\label{ext_fibration_existence}
\lean{S4TuringPoisson.closedManifoldRegularDiskFibration,
  S4TuringPoisson.AdmitsRegularDiskFibration.ofPositiveVolumeForm,
  S4TuringPoisson.not_isGenericallyRankTwo_closedManifoldLocal,
  S4TuringPoisson.not_admitsWrinkledOrBrokenLefschetzFibration_closedManifoldLocal}
\leanok
\uses{def_admits, def_s4_fibration}
Every nonempty compact Hausdorff smooth $4$-manifold $M$ equipped with a positive volume $4$-form
$V_M$ admits a positively co-oriented \texttt{RegularDiskFibration} $M\to S^2$ with fiber $S^2$
(\texttt{ofPositiveVolumeForm}, Track~B), constructed locally inside an orientation-preserving
coordinate ball and constant outside it (hence proved \emph{not} generically rank $2$ and unable to
inhabit \texttt{AdmitsWrinkledOrBrokenLefschetzFibration}).
\end{theorem}
\begin{proof}\leanok
Compose an orientation-preserving coordinate-ball chart into $M$ with a radial cut-off blend of
\texttt{s4Fibration} collapsing the exterior to a single point outside \texttt{s4BaseRegion};
compactness of $M$ implies the exterior has nonempty interior inside $\Crit(f)$.
\end{proof}

\begin{theorem}[Componentwise Moser volume conjugation on compact $4$-manifolds]
\label{thm_moser_volume}
\lean{S4TuringPoisson.MoserVolume, S4TuringPoisson.MoserVolume.ofPositivelyCooriented}
\leanok
\uses{def_volume, def_jacobian}
For any two positively co-oriented smooth positive volume forms $V_0,V_1$ on a nonempty compact
Hausdorff smooth $4$-manifold $M$ (without assuming connectedness), there is a diffeomorphism
$\psi\colon M\simeq_m M$ preserving each connected component $C$ with $\psi^*V_1|_C = c_C\cdot V_0|_C$
($c_C>0$); conjugating any component-preserving $V_1$-preserving diffeomorphism $\dbar$ to
$d_{\mathrm{exact}}=\psi^{-1}\circ\dbar\circ\psi$ preserves $V_0.\Omega$ and $V_0.\mu$ identically.
\end{theorem}
\begin{proof}\leanok
Apply TauCeti's top-degree de Rham integration isomorphism and time-dependent Moser flow on each
clopen connected component of $M$.
\end{proof}

\end{theorem}

\end{proof}

\begin{theorem}[Turing completeness of the lift]
\label{thm_lift_tc}
\lean{S4TuringPoisson.leafwiseBumpDiffeomorph, S4TuringPoisson.isTuringComplete_leafwiseBumpDiffeomorph,

\end{theorem}

\end{proof}

\begin{definition}[Universal reversible simulator and the CMPP class]

\begin{lemma}[Simulator to witness]
\label{lem_sim_witness}
\lean{S4TuringPoisson.UniversalReversibleSimulator.turingCompleteWitness,
  S4TuringPoisson.instHasTuringCompleteAreaPreservingDiskIsotopyOfCMPP}
\leanok
\uses{def_simulator, def_turing_complete}
A map conjugate to the simulator's shift on the Cantor set is Turing complete; hence the CMPP
class yields a Turing complete area-preserving disk isotopy.
\end{lemma}
\begin{proof}\leanok
